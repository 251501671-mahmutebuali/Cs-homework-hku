using System.Diagnostics.CodeAnalysis;
using System.Security.Cryptography.X509Certificates;

Console.Write("Type a grade: ");
int grade = Convert.ToInt32(Console.ReadLine());

if (grade >= 0 && grade <= 100)
{
    if (grade >= 90 && grade <= 100)
    {
        Console.WriteLine($"Score: {grade} --> A");
    }
    else if (grade >= 80 && grade <= 89)
    {
        Console.WriteLine($"Score: {grade} --> B");
    }
    else if (grade >= 70 && grade <= 79)
    {
        Console.WriteLine($"Score: {grade} --> C");
    }
    else if (grade >= 60 && grade <= 69)
    {
        Console.WriteLine($"Score: {grade} --> D");
    }
    else if (grade < 60)
    {
        Console.WriteLine($"Score: {grade} --> F");
    }
}

int x = 1;
int y = 0;
int sum = 0;
while (x > 0)
{
    Console.Write("Number (0 to stop): ");
    x = Convert.ToInt32(Console.ReadLine());
    if (x == 0)
    {
        break;
    }

    sum += x;
    y += 1;

}

if (y > 0)
{
    double average = (double)sum / y;

    Console.WriteLine($"Count: {y} Sum: {sum} Average: {average}");
}
